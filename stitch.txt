        function toggleGoal(element) {
            // Check if it's already active
            const isActive = element.classList.contains('goal-card-active');
            
            // For a single choice behavior, uncomment next 3 lines:

            if (!isActive) {

                element.querySelector('.check-icon').classList.remove('opacity-0');
                element.querySelector('.check-icon').classList.add('opacity-100');

            }

                vatRadio.classList.add('border-outline-variant');
                vatRadio.classList.remove('border-primary-container');

                simpleRadio.classList.add('border-outline-variant');
                simpleRadio.classList.remove('border-primary-container');

        // Simple micro-interaction for the bottom sheet
        const sheet = document.querySelector('.bottom-sheet');
        let isDragging = false;
        let startY = 0;

        sheet.addEventListener('touchstart', (e) => {
            startY = e.touches[0].clientY;
            isDragging = true;
        });

        sheet.addEventListener('touchmove', (e) => {
            if (!isDragging) return;
            const currentY = e.touches[0].clientY;
            const diff = currentY - startY;
            if (diff > 0) {
                sheet.style.transform = `translateY(${diff}px)`;
            }
        });

        sheet.addEventListener('touchend', (e) => {
            isDragging = false;
            sheet.style.transform = `translateY(0px)`;
        });
        
        // Dynamic Driver Movement Simulation
        const driverDot = document.querySelector('.absolute.top-\\[320px\\]');
        let moveOffset = 0;
        setInterval(() => {
            moveOffset += 0.2;
            const x = 270 + Math.sin(moveOffset) * 10;
            const y = 320 + Math.cos(moveOffset) * 10;
            driverDot.style.left = `${x}px`;
            driverDot.style.top = `${y}px`;
        }, 100);
