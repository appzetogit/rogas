1. App Overview & Architecture 
 
The DailyMealBox Driver App guides drivers through their entire shift: from going online, receiving their optimised multi-stop route, collecting boxes from 
vendors (with PIN verification), delivering to customers (with PIN or photo proof), and managing their earnings and cash COD balance. All delivery logic 
runs on the backend — driver app only displays and navigates.

// Driver taps Go Online:​
POST /api/drivers/{id}/online​
​
Backend validation:​
1. Check shift is active:​
   const shiftStart = remoteConfig.get("lunch_shift_start_warsaw") // "11:00"​
   const shiftEnd   = remoteConfig.get("lunch_shift_end_warsaw")   // "14:00"​
   if (now < shiftStart || now > shiftEnd)​
     return error("Your lunch shift starts at " + shiftStart)​
​
2. Check driver active + documents not expired:​
   if (driver.status !== "active")​
     return error({ code: "account_suspended" })​

3. Set online + start GPS streaming:​
   driver.is_online = true​
   redis.set("driver:" + driver.id + ":online", 1, "EX", 18000) // 5h max​

Delivery Confirmation — PIN or Photo + Fraud Check 
// Method A — PIN:​
POST /api/delivery/verify-pin​
{ order_id, pin_entered: "7234", driver_id, delivery_gps: {lat, lng} }​
​
Backend:​
  const stored = redis.get("delivery_pin:" + order_id)​
  if (pin_entered !== stored) return error("Incorrect PIN")​
​
  // FRAUD CHECK (v3.0):​
  const distance = haversineDistance(delivery_gps, order.delivery_address)​
  if (distance > 500) {​
    flagForReview(driver_id, "GPS_MISMATCH", { distance, order_id })​
    // Still confirm delivery (driver may be in alley) but flag for admin​
  }​
  confirmDelivery(order_id, driver_id, "pin", delivery_gps)​
​
// Method B — Photo:​
POST /api/delivery/verify-photo (multipart)​
{ order_id, photo: imageFile, driver_id, delivery_gps: {lat, lng} }​
  photo_url = await uploadPhoto(imageFile) // Firebase Storage​
  confirmDelivery(order_id, driver_id, "photo", delivery_gps, photo_url)​
​
// confirmDelivery — shared function:​
async function confirmDelivery(order_id, driver_id, method, gps, photo=null) {​
  // Update order:​
  order.status = "delivered"; order.delivered_at = now()​
  order.proof_method = method; order.proof_photo_url = photo​
  order.delivery_gps = gps​
​
  // COD cash (v3.0 — fleet partner owns tip VAT):​
  if (order.payment_method === "cod") {​
    driver.cash_balance += order.price_pln​
    if (driver.cash_balance >= driver.cash_limit_pln) {​
      FCM.send(driver, "Cash limit reached — contact admin")​
      notifyAdmin("CASH_LIMIT", driver)​
    }​
  }​
​
  // Tip VAT note (v3.0 correction):​
  // Tip gross amount goes 100% to driver via fleet partner​
  // Fleet partner declares 23% VAT on tips to tax authority​
  // Driver sees tip gross amount in earnings (VAT already handled by fleet)​
​
  FCM.send(customer, "Delivered! Enjoy your meal. Rate your experience →")​
  driver.earnings_today += order.delivery_fee_pln​
  driver.deliveries_today += 1​
  checkBonusProgress(driver_id)​
}
