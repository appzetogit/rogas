the system splits VAT ownership: food VAT belongs to vendor, delivery VAT belongs to fleet partner, commission VAT belongs to DailyMealBox.

VAT Engine — Core Calculation (Node.js) 
// VAT rates from Remote Config (admin sets per city — OTA, no reinstall):​
const VAT_RATES = {​
  warsaw: { food_restaurant:0.08, food_basic:0.05, delivery:0.23, service:0.23, tips:0.23 },​
  berlin: { food_restaurant:0.07, food_basic:0.07, delivery:0.19, service:0.19, tips:0.19 },​
  paris:  { food_restaurant:0.10, food_basic:0.055, delivery:0.20, service:0.20, tips:0.20 }​
}​
​
function calculateOrderVAT(order, city, customerType, customerVatNumber) {​
  const rates = VAT_RATES[city]​
​
  // EU reverse charge: B2B buyer from different EU country:​
  const isReverseCharge = (customerType==="b2b" && customerVatNumber &&​
    getEuCountry(customerVatNumber) !== getEuCountry(city))​
​
  const foodRate = isReverseCharge ? 0 :​
    (order.vendor.food_category==="basic" ? rates.food_basic : rates.food_restaurant)​
  const deliveryRate = isReverseCharge ? 0 : rates.delivery​
​
  const foodNet = order.meal_price / (1 + foodRate)​
  const foodVat = order.meal_price - foodNet​
  const deliveryNet = order.delivery_fee / (1 + deliveryRate)​
  const deliveryVat = order.delivery_fee - deliveryNet​
​
  return {​
    // Customer sees:​
    total_gross: order.meal_price + order.delivery_fee + (order.tip || 0),​
​
    // Invoice breakdown:​
    food_net: foodNet.toFixed(2), food_vat_rate: foodRate*100+"%",​
    food_vat_amount: foodVat.toFixed(2), food_vat_owner: "vendor",​
​
    delivery_net: deliveryNet.toFixed(2), delivery_vat_rate: deliveryRate*100+"%",​
    delivery_vat_amount: deliveryVat.toFixed(2), delivery_vat_owner: "fleet_partner",​
​
    is_reverse_charge: isReverseCharge,​
    reverse_charge_notice: isReverseCharge ?​
      "Reverse charge — Art. 196 EU VAT Directive" : null,​
​
    // Internal settlement:​
    vendor_receives_net: foodNet,​
    fleet_receives_net: deliveryNet,​
    platform_commission: foodNet * order.vendor.commission_rate,​
  }​
}

3. Home Cook Legal Model — Kitchen Partner Entity  
 
Home cooks in Poland often lack a registered company or VAT number. Without a legal entity they cannot issue food invoices or receive commercial bank 
payments. Solution: Home Cook links to a Kitchen Partner company — a registered Sp. z o.o. that handles all legal, invoicing, and financial aspects on their 
behalf. This mirrors the model used by catering platforms across Poland.

Calendar Cutoff Lock Logic 
function getDayState(order, vendorSettings) {​
  const now = new Date()​
  const cutoff = calculateCutoff(order.delivery_date, vendorSettings)​
  if (now > cutoff) return { state:"locked", icon:"🔒", actions:[] }​
  const skipsThisMonth = getSkipsThisMonth(user.id)​
  const maxSkips = remoteConfig.getNumber("max_skips_per_month") || 2​
  return {​
    state: "scheduled",​
    actions: [​
      skipsThisMonth < maxSkips ? "skip" : "skip_disabled",​
      "reschedule", "swap_meal"​
    ]​
  }​
}​

// Runs 1st of each month at 06:00 for previous month:​
cron.schedule("0 6 1 * *", async () => {​
  const companies = await B2BAccount.find({ status: "active" })​
  for (const company of companies) {​
    const orders = await Order.find({​
      company_id: company._id,​
      delivery_date: { $gte: startOfLastMonth(), $lte: endOfLastMonth() },​
      status: "delivered"​
    })​
    const invoiceNumber = await getNextInvoiceNumber(company.city)​
    const isReverseCharge = isEUCrossBorder(company.vat_number, company.city)​
    const foodRate = isReverseCharge ? 0 : VAT_RATES[company.city].food_restaurant​
    const deliveryRate = isReverseCharge ? 0 : VAT_RATES[company.city].delivery​
​
    const pdfBuffer = await generateInvoicePDF({​
      invoice_number: invoiceNumber,​
      seller: { name:"DailyMealBox Sp. z o.o.", nip: PLATFORM_NIP },​
      buyer: { name: company.name, nip: company.vat_number,​
               address: company.billing_address },​
      line_items: buildLineItems(orders, foodRate, deliveryRate),​
      reverse_charge: isReverseCharge,​
      payment_terms: "14 days bank transfer"​
    })​
​
    const pdfUrl = await uploadToFirebase("/invoices/b2b/"+invoiceNumber+".pdf", pdfBuffer)​
    await email.send(company.billing_email, "Invoice "+invoiceNumber, { pdfUrl })​
    await Invoice.create({ invoice_number:invoiceNumber, company_id:company._id,​
      pdf_url:pdfUrl, status:"sent", retention_until: addYears(now(),7) })​
  }​
}, { timezone: "Europe/Warsaw" })
